-- ==========================================================
-- Part 1: Data Definition Language (DDL)
-- ==========================================================

-- 1. Database Creation
CREATE DATABASE IF NOT EXISTS `Pine Valley Furniture`;
USE `Pine Valley Furniture`;

-- 2. Create Tables (Parent Tables First)
CREATE TABLE Customer_T (
    CustomerID INT PRIMARY KEY,
    CustomerName VARCHAR(100) NOT NULL,
    CustomerAddress VARCHAR(150),
    CustomerCity VARCHAR(50),
    CustomerState VARCHAR(50),
    CustomerPostalCode VARCHAR(20)
);

CREATE TABLE Product_T (
    ProductID INT PRIMARY KEY,
    ProductDescription VARCHAR(100),
    ProductFinish VARCHAR(50),
    ProductStandardPrice DECIMAL(10,2),
    CONSTRAINT chk_ProductFinish CHECK (ProductFinish IN ('Natural Ash', 'Red Oak', 'Cherry', 'Walnut'))
);

-- Child Tables (Referencing Parent Tables)
CREATE TABLE Order_T (
    OrderID INT PRIMARY KEY,
    OrderDate DATE,
    CustomerID INT,
    FOREIGN KEY (CustomerID) REFERENCES Customer_T(CustomerID) ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE OrderLine_T (
    OrderID INT,
    ProductID INT,
    OrderedQuantity INT,
    PRIMARY KEY (OrderID, ProductID),
    FOREIGN KEY (OrderID) REFERENCES Order_T(OrderID) ON DELETE RESTRICT ON UPDATE CASCADE,
    FOREIGN KEY (ProductID) REFERENCES Product_T(ProductID) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- 3. Schema Modification
ALTER TABLE Customer_T ADD CustomerEmail VARCHAR(50) UNIQUE;

-- 4. Create Index for performance
CREATE INDEX idx_product_price ON Product_T(ProductStandardPrice);

-- (b) Aggregate Functions
SELECT MAX(ProductStandardPrice) AS Max_Price,
       MIN(ProductStandardPrice) AS Min_Price,
       AVG(ProductStandardPrice) AS Avg_Price
FROM Product_T;

INNER JOIN Customer_T ON Order_T.CustomerID = Customer_T.CustomerID;
